cmake_minimum_required(VERSION 3.20)
project(metamaterial_compression LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(metamaterial_compression src/main.cpp src/sim.cpp src/render.cpp)
target_include_directories(metamaterial_compression PRIVATE include)
if(MSVC)
  target_compile_options(metamaterial_compression PRIVATE /O2 /W4 /permissive-)
else()
  target_compile_options(metamaterial_compression PRIVATE -O3 -Wall -Wextra -Wpedantic)
  target_link_libraries(metamaterial_compression PRIVATE pthread)
endif()
include(CTest)
if(BUILD_TESTING)
  add_test(NAME metamaterial_numerical_validation COMMAND metamaterial_compression --mode validate --quick --output ${CMAKE_BINARY_DIR}/validation)
endif()
