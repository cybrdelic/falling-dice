cmake_minimum_required(VERSION 3.20)
project(metamaterial_fem LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(OpenMP)

add_library(meta_fem STATIC src/fem.cpp)
target_include_directories(meta_fem PUBLIC include)

add_library(meta_render STATIC src/render.cpp)
target_include_directories(meta_render PUBLIC include)
target_link_libraries(meta_render PUBLIC meta_fem)

add_executable(metamaterial_fem src/main.cpp)
target_link_libraries(metamaterial_fem PRIVATE meta_fem meta_render)

include(CTest)
if(BUILD_TESTING)
  add_executable(metamaterial_verification tests/verification.cpp)
  target_link_libraries(metamaterial_verification PRIVATE meta_fem)
  add_test(NAME metamaterial_verification COMMAND metamaterial_verification)
endif()

foreach(target meta_fem meta_render metamaterial_fem)
  if(OpenMP_CXX_FOUND)
    target_link_libraries(${target} PUBLIC OpenMP::OpenMP_CXX)
  endif()
  if(MSVC)
    target_compile_options(${target} PRIVATE /O2 /W4 /permissive-)
  else()
    target_compile_options(${target} PRIVATE -O3 -Wall -Wextra -Wpedantic)
    target_link_libraries(${target} PUBLIC pthread)
  endif()
endforeach()

if(TARGET metamaterial_verification)
  if(OpenMP_CXX_FOUND)
    target_link_libraries(metamaterial_verification PRIVATE OpenMP::OpenMP_CXX)
  endif()
  if(MSVC)
    target_compile_options(metamaterial_verification PRIVATE /O2 /W4 /permissive-)
  else()
    target_compile_options(metamaterial_verification PRIVATE -O3 -Wall -Wextra -Wpedantic)
    target_link_libraries(metamaterial_verification PRIVATE pthread)
  endif()
endif()
